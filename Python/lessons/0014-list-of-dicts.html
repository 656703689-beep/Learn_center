02 · 新知识 · 4 分钟

list 的元素可以是 dict

reports = [
    {"name": "第一组", "hits": 1},
    {"name": "第二组", "hits": 0}
]

for report in reports:
    print(report["name"])
    print(report["hits"])

reports 是一个 list。这个 list 里有两个元素，每个元素都是一个 dict。

一句话抓住今天 外层 list 负责“多份”；内层 dict 负责“每份里有什么字段”。

依据：Python 官方教程：列表和 Python 官方教程：字典。本课把两个已学结构组合使用。

04 · 实作 · 8 分钟

把两组猜测都变成报告

在 Python 文件夹中新建 multi_reports.py。这次我们先生成两份报告，再把它们放进 reports 列表。

def compare_guess(guess, secret):
    if guess < secret:
        return "太小了"
    elif guess > secret:
        return "太大了"
    else:
        return "猜中了"


def suggest_from_counts(small_count, big_count, hit_count):
    if hit_count > 0:
        return "已经猜中过，测试通过"
    elif small_count > big_count:
        return "整体偏小，可以试大一点"
    elif big_count > small_count:
        return "整体偏大，可以试小一点"
    else:
        return "偏小和偏大一样多"


def build_report(name, guesses, secret):
    results = []

    for guess in guesses:
        result = compare_guess(guess, secret)
        results.append(result)

    small_count = results.count("太小了")
    big_count = results.count("太大了")
    hit_count = results.count("猜中了")
    summary = suggest_from_counts(small_count, big_count, hit_count)

    return {
        "name": name,
        "total": len(results),
        "small": small_count,
        "big": big_count,
        "hits": hit_count,
        "summary": summary
    }


secret = 30

report1 = build_report("第一组", [10, 30, 40], secret)
report2 = build_report("第二组", [35, 40, 20], secret)

reports = [report1, report2]

for report in reports:
    print(report["name"])
    print("总次数：", report["total"])
    print("猜中次数：", report["hits"])
    print("总结：", report["summary"])
